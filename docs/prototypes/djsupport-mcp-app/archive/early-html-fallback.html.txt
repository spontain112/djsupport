    function setVariant(key){variant=key;document.body.className='variant-'+key.toLowerCase();document.getElementById('variantLabel').textContent=key+' — '+variants.find(v=>v[0]===key)[1];params.set('variant',key);history.replaceState(null,'','?'+params.toString()+location.hash)}
    function cycle(delta){const i=variants.findIndex(v=>v[0]===variant);setVariant(variants[(i+delta+variants.length)%variants.length][0])}
    function showScreen(id){document.querySelectorAll('.screen').forEach(s=>s.classList.toggle('active',s.id===id));document.querySelectorAll('.nav button').forEach(b=>b.classList.toggle('active',b.dataset.screen===id));location.hash=id;document.querySelector('.main').scrollTop=0}
    document.querySelectorAll('[data-screen]').forEach(b=>b.onclick=()=>showScreen(b.dataset.screen));document.querySelectorAll('[data-next]').forEach(b=>b.onclick=()=>showScreen(b.dataset.next));
    document.querySelectorAll('[data-action]').forEach(b=>b.addEventListener('click',()=>{const t=document.querySelector('.toast');t.textContent=b.dataset.action==='approve'?'Approval recorded — no playlist mutation':'Prototype state updated';t.classList.add('show');setTimeout(()=>t.classList.remove('show'),1400)}));
    document.querySelectorAll('.choice').forEach(b=>b.onclick=()=>{document.querySelectorAll('.choice').forEach(x=>x.classList.remove('selected'));b.classList.add('selected')});
    document.getElementById('prev').onclick=()=>cycle(-1);document.getElementById('next').onclick=()=>cycle(1);document.addEventListener('keydown',e=>{if(['INPUT','TEXTAREA'].includes(document.activeElement.tagName)||document.activeElement.isContentEditable)return;if(e.key==='ArrowLeft')cycle(-1);if(e.key==='ArrowRight')cycle(1)});
    setVariant(variant);showScreen(location.hash.slice(1)||'start');
